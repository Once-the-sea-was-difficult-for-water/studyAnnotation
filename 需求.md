# 爬虫平台一期：MySQL DDL 与 API 清单（草案）

> 目标：提供可直接进入研发拆解的最小可用设计。  
> 技术栈约束：`Spring Boot`、`MySQL`、`MongoDB`、`Vue`、`Formily`、`OSS`、`Supervisor`。

---

## 1. MySQL DDL（一期最小集）

> 约定：字符集 `utf8mb4`，存储引擎 `InnoDB`，时间字段统一 `datetime`。  
> 说明：原始采集内容和大体量日志落 Mongo，此处保留索引与统计字段。

```sql
CREATE DATABASE IF NOT EXISTS crawler_platform
  DEFAULT CHARACTER SET utf8mb4
  DEFAULT COLLATE utf8mb4_general_ci;

USE crawler_platform;

-- 1) 模板定义（Formily schema）
CREATE TABLE IF NOT EXISTS template (
  id BIGINT PRIMARY KEY AUTO_INCREMENT,
  template_code VARCHAR(64) NOT NULL UNIQUE COMMENT '模板唯一编码',
  template_name VARCHAR(128) NOT NULL COMMENT '模板名称',
  version INT NOT NULL DEFAULT 1 COMMENT '模板版本',
  form_schema JSON NOT NULL COMMENT 'Formily schema',
  status TINYINT NOT NULL DEFAULT 1 COMMENT '1=启用,0=停用',
  remark VARCHAR(500) DEFAULT NULL,
  created_by VARCHAR(64) NOT NULL,
  updated_by VARCHAR(64) NOT NULL,
  created_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
  updated_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
  INDEX idx_template_name (template_name),
  INDEX idx_template_status (status)
) COMMENT='采集任务模板';

-- 2) 脚本主表
CREATE TABLE IF NOT EXISTS script (
  id BIGINT PRIMARY KEY AUTO_INCREMENT,
  script_code VARCHAR(64) NOT NULL UNIQUE COMMENT '脚本唯一编码',
  script_name VARCHAR(128) NOT NULL,
  language VARCHAR(16) NOT NULL COMMENT 'python/node',
  owner VARCHAR(64) NOT NULL COMMENT '负责人',
  status TINYINT NOT NULL DEFAULT 1 COMMENT '1=可用,0=停用',
  remark VARCHAR(500) DEFAULT NULL,
  created_by VARCHAR(64) NOT NULL,
  updated_by VARCHAR(64) NOT NULL,
  created_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
  updated_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
  INDEX idx_script_name (script_name),
  INDEX idx_script_lang (language),
  INDEX idx_script_status (status)
) COMMENT='脚本主信息';

-- 3) 脚本版本表（OSS + 校验）
CREATE TABLE IF NOT EXISTS script_version (
  id BIGINT PRIMARY KEY AUTO_INCREMENT,
  script_id BIGINT NOT NULL,
  version_no VARCHAR(32) NOT NULL COMMENT '如 v1.0.3',
  oss_bucket VARCHAR(64) NOT NULL,
  oss_object_key VARCHAR(255) NOT NULL,
  file_sha256 VARCHAR(64) NOT NULL,
  entry_file VARCHAR(128) NOT NULL COMMENT '入口文件',
  runtime_args VARCHAR(500) DEFAULT NULL COMMENT '默认启动参数',
  publish_status TINYINT NOT NULL DEFAULT 0 COMMENT '0=草稿,1=测试,2=已发布',
  changelog VARCHAR(1000) DEFAULT NULL,
  created_by VARCHAR(64) NOT NULL,
  created_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
  UNIQUE KEY uk_script_ver (script_id, version_no),
  INDEX idx_script_publish (script_id, publish_status),
  CONSTRAINT fk_script_version_script FOREIGN KEY (script_id) REFERENCES script(id)
) COMMENT='脚本版本';

-- 4) 依赖包定义（python/node）
CREATE TABLE IF NOT EXISTS dependency_package (
  id BIGINT PRIMARY KEY AUTO_INCREMENT,
  dep_code VARCHAR(64) NOT NULL UNIQUE COMMENT '依赖包唯一编码',
  dep_name VARCHAR(128) NOT NULL,
  language VARCHAR(16) NOT NULL COMMENT 'python/node',
  lock_content MEDIUMTEXT NOT NULL COMMENT 'requirements.txt 或 package-lock.json内容',
  lock_sha256 VARCHAR(64) NOT NULL,
  install_strategy TINYINT NOT NULL DEFAULT 1 COMMENT '1=在线安装,2=离线包',
  offline_oss_bucket VARCHAR(64) DEFAULT NULL,
  offline_oss_key VARCHAR(255) DEFAULT NULL,
  status TINYINT NOT NULL DEFAULT 1 COMMENT '1=可用,0=停用',
  created_by VARCHAR(64) NOT NULL,
  updated_by VARCHAR(64) NOT NULL,
  created_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
  updated_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
  INDEX idx_dep_name (dep_name),
  INDEX idx_dep_lang (language),
  INDEX idx_dep_status (status)
) COMMENT='依赖包定义';

-- 5) 任务定义（模板实例化）
CREATE TABLE IF NOT EXISTS task (
  id BIGINT PRIMARY KEY AUTO_INCREMENT,
  task_code VARCHAR(64) NOT NULL UNIQUE,
  task_name VARCHAR(128) NOT NULL,
  template_id BIGINT NOT NULL,
  template_version INT NOT NULL,
  template_snapshot JSON NOT NULL COMMENT '任务创建时模板快照',
  form_data JSON NOT NULL COMMENT '用户填写参数',
  script_id BIGINT NOT NULL,
  script_version_id BIGINT NOT NULL,
  dependency_id BIGINT NOT NULL,
  site_domain VARCHAR(128) DEFAULT NULL COMMENT '目标域名，用于限流分组',
  priority INT NOT NULL DEFAULT 5 COMMENT '1最高,10最低',
  timeout_sec INT NOT NULL DEFAULT 600,
  max_retry INT NOT NULL DEFAULT 2,
  status TINYINT NOT NULL DEFAULT 1 COMMENT '1=启用,0=停用',
  created_by VARCHAR(64) NOT NULL,
  updated_by VARCHAR(64) NOT NULL,
  created_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
  updated_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
  INDEX idx_task_name (task_name),
  INDEX idx_task_status (status),
  INDEX idx_task_domain (site_domain),
  CONSTRAINT fk_task_template FOREIGN KEY (template_id) REFERENCES template(id),
  CONSTRAINT fk_task_script FOREIGN KEY (script_id) REFERENCES script(id),
  CONSTRAINT fk_task_script_ver FOREIGN KEY (script_version_id) REFERENCES script_version(id),
  CONSTRAINT fk_task_dep FOREIGN KEY (dependency_id) REFERENCES dependency_package(id)
) COMMENT='采集任务定义';

-- 6) 调度策略
CREATE TABLE IF NOT EXISTS task_schedule (
  id BIGINT PRIMARY KEY AUTO_INCREMENT,
  task_id BIGINT NOT NULL UNIQUE,
  schedule_type TINYINT NOT NULL COMMENT '1=手动,2=立即一次,3=cron',
  cron_expr VARCHAR(64) DEFAULT NULL,
  timezone VARCHAR(64) DEFAULT 'Asia/Shanghai',
  next_trigger_time DATETIME DEFAULT NULL,
  status TINYINT NOT NULL DEFAULT 1 COMMENT '1=启用,0=停用',
  updated_by VARCHAR(64) NOT NULL,
  created_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
  updated_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
  INDEX idx_schedule_next (next_trigger_time),
  INDEX idx_schedule_status (status),
  CONSTRAINT fk_schedule_task FOREIGN KEY (task_id) REFERENCES task(id)
) COMMENT='任务调度配置';

-- 7) Worker 节点注册
CREATE TABLE IF NOT EXISTS worker_node (
  id BIGINT PRIMARY KEY AUTO_INCREMENT,
  node_code VARCHAR(64) NOT NULL UNIQUE,
  node_name VARCHAR(128) NOT NULL,
  ip VARCHAR(64) NOT NULL,
  tags VARCHAR(255) DEFAULT NULL COMMENT '逗号分隔标签，如 cn-east,python',
  capacity INT NOT NULL DEFAULT 4 COMMENT '最大并发槽位',
  used_slots INT NOT NULL DEFAULT 0,
  status TINYINT NOT NULL DEFAULT 1 COMMENT '1=在线,0=离线,2=维护',
  heartbeat_at DATETIME DEFAULT NULL,
  supervisor_version VARCHAR(32) DEFAULT NULL,
  agent_version VARCHAR(32) DEFAULT NULL,
  created_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
  updated_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
  INDEX idx_node_status (status),
  INDEX idx_node_heartbeat (heartbeat_at)
) COMMENT='工作节点';

-- 8) 任务实例（一次执行）
CREATE TABLE IF NOT EXISTS task_instance (
  id BIGINT PRIMARY KEY AUTO_INCREMENT,
  instance_code VARCHAR(64) NOT NULL UNIQUE,
  task_id BIGINT NOT NULL,
  trigger_type TINYINT NOT NULL COMMENT '1=调度触发,2=手工触发,3=重试触发',
  status TINYINT NOT NULL DEFAULT 0 COMMENT '0=待执行,1=运行中,2=成功,3=失败,4=取消,5=超时',
  retry_no INT NOT NULL DEFAULT 0,
  worker_node_id BIGINT DEFAULT NULL,
  queue_time DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
  start_time DATETIME DEFAULT NULL,
  end_time DATETIME DEFAULT NULL,
  duration_ms BIGINT DEFAULT NULL,
  exit_code INT DEFAULT NULL,
  error_code VARCHAR(64) DEFAULT NULL,
  error_message VARCHAR(1000) DEFAULT NULL,
  mongo_data_ref VARCHAR(128) DEFAULT NULL COMMENT 'Mongo 数据主键/分片键',
  log_ref VARCHAR(128) DEFAULT NULL COMMENT '日志引用',
  created_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
  updated_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
  INDEX idx_inst_task (task_id),
  INDEX idx_inst_status (status),
  INDEX idx_inst_worker (worker_node_id),
  INDEX idx_inst_queue_time (queue_time),
  CONSTRAINT fk_inst_task FOREIGN KEY (task_id) REFERENCES task(id),
  CONSTRAINT fk_inst_worker FOREIGN KEY (worker_node_id) REFERENCES worker_node(id)
) COMMENT='任务执行实例';

-- 9) 实例日志索引（日志主体在 Mongo）
CREATE TABLE IF NOT EXISTS task_instance_log_index (
  id BIGINT PRIMARY KEY AUTO_INCREMENT,
  task_instance_id BIGINT NOT NULL,
  log_type TINYINT NOT NULL COMMENT '1=stdout,2=stderr,3=system',
  mongo_log_ref VARCHAR(128) NOT NULL,
  line_count INT NOT NULL DEFAULT 0,
  created_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
  INDEX idx_log_inst (task_instance_id),
  INDEX idx_log_type (log_type),
  CONSTRAINT fk_log_inst FOREIGN KEY (task_instance_id) REFERENCES task_instance(id)
) COMMENT='实例日志索引';

-- 10) 操作审计
CREATE TABLE IF NOT EXISTS operation_audit (
  id BIGINT PRIMARY KEY AUTO_INCREMENT,
  operator VARCHAR(64) NOT NULL,
  operation_type VARCHAR(64) NOT NULL COMMENT 'CREATE_TASK/PUBLISH_SCRIPT/RETRY_INSTANCE...',
  biz_type VARCHAR(64) NOT NULL COMMENT 'task/script/template/node',
  biz_id VARCHAR(64) NOT NULL,
  request_payload JSON DEFAULT NULL,
  result_status TINYINT NOT NULL COMMENT '1=成功,0=失败',
  result_message VARCHAR(500) DEFAULT NULL,
  created_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
  INDEX idx_audit_operator (operator),
  INDEX idx_audit_biz (biz_type, biz_id),
  INDEX idx_audit_time (created_at)
) COMMENT='操作审计日志';
```

## 2. 状态枚举与字段约束建议

- `task.status`：`1=启用`，`0=停用`
- `task_instance.status`：`0待执行/1运行中/2成功/3失败/4取消/5超时`
- `script_version.publish_status`：`0草稿/1测试/2已发布`
- `worker_node.status`：`1在线/0离线/2维护`
- 统一要求：
  - 核心业务表必须有 `created_at/updated_at`
  - 外部可见编码字段统一 `*_code` 并加唯一索引
  - JSON 字段仅承载可变配置，查询条件字段需冗余为结构化列

## 3. API 清单（Spring Boot，`/api/v1`）

> 建议统一响应：
> - 成功：`{ code: 0, message: "ok", data: ... }`
> - 失败：`{ code: <业务码>, message: <错误信息>, data: null }`

## 3.1 模板管理

- `POST /templates`：创建模板
- `PUT /templates/{id}`：更新模板
- `GET /templates/{id}`：模板详情
- `GET /templates`：模板分页查询
- `POST /templates/{id}/disable`：停用模板
- `POST /templates/{id}/enable`：启用模板

请求示例（创建模板）：

```json
{
  "templateCode": "TMP_NEWS_SITE",
  "templateName": "新闻站点采集模板",
  "formSchema": {},
  "remark": "一期默认模板"
}
```

## 3.2 脚本与版本

- `POST /scripts`：创建脚本
- `GET /scripts`：脚本列表
- `GET /scripts/{id}`：脚本详情
- `POST /scripts/{id}/versions`：上传新版本（记录 OSS 路径）
- `GET /scripts/{id}/versions`：版本列表
- `POST /script-versions/{id}/publish`：发布版本
- `POST /script-versions/{id}/test-run`：测试运行（指定测试参数）

请求示例（新增脚本版本）：

```json
{
  "versionNo": "v1.0.0",
  "ossBucket": "crawler-script-prod",
  "ossObjectKey": "scripts/news/v1.0.0.zip",
  "fileSha256": "xxxx",
  "entryFile": "main.py",
  "runtimeArgs": "--env=prod"
}
```

## 3.3 依赖包管理

- `POST /dependencies`：创建依赖包
- `PUT /dependencies/{id}`：更新依赖包
- `GET /dependencies/{id}`：依赖详情
- `GET /dependencies`：依赖列表（按语言筛选）
- `POST /dependencies/{id}/disable`：停用
- `POST /dependencies/{id}/enable`：启用

## 3.4 任务管理

- `POST /tasks`：创建任务
- `PUT /tasks/{id}`：更新任务
- `GET /tasks/{id}`：任务详情
- `GET /tasks`：任务分页查询
- `POST /tasks/{id}/enable`：启用任务
- `POST /tasks/{id}/disable`：停用任务
- `POST /tasks/{id}/run-once`：立即执行一次
- `POST /tasks/{id}/retry-last-failed`：重试最近失败实例

请求示例（创建任务）：

```json
{
  "taskCode": "TASK_NEWS_001",
  "taskName": "新闻采集任务-001",
  "templateId": 1,
  "formData": {
    "entryUrl": "https://example.com",
    "pageLimit": 50
  },
  "scriptId": 1,
  "scriptVersionId": 3,
  "dependencyId": 2,
  "siteDomain": "example.com",
  "priority": 3,
  "timeoutSec": 900,
  "maxRetry": 2
}
```

## 3.5 调度管理

- `PUT /tasks/{id}/schedule`：创建/更新调度策略
- `GET /tasks/{id}/schedule`：调度详情
- `POST /tasks/{id}/schedule/enable`：启用调度
- `POST /tasks/{id}/schedule/disable`：停用调度

请求示例（设置 cron）：

```json
{
  "scheduleType": 3,
  "cronExpr": "0 */30 * * * ?",
  "timezone": "Asia/Shanghai"
}
```

## 3.6 执行实例与日志

- `GET /task-instances`：实例分页查询（支持按任务、状态、时间）
- `GET /task-instances/{id}`：实例详情
- `POST /task-instances/{id}/cancel`：取消运行中实例
- `GET /task-instances/{id}/logs`：拉取日志（分页/增量）
- `GET /task-instances/{id}/result`：获取结果摘要与数据引用

## 3.7 Worker 节点管理

- `POST /workers/register`：节点注册
- `POST /workers/{nodeCode}/heartbeat`：心跳上报
- `GET /workers`：节点列表
- `POST /workers/{id}/offline`：手动下线
- `POST /workers/{id}/online`：手动上线

Agent 通信接口（内部）：
- `POST /internal/dispatch/pull`：Worker 拉取任务
- `POST /internal/dispatch/report-status`：上报状态
- `POST /internal/dispatch/report-log`：上报日志索引

## 3.8 审计与监控

- `GET /audits`：操作审计查询
- `GET /metrics/overview`：平台总览（成功率、失败率、运行中数）
- `GET /metrics/tasks/{taskId}`：任务维度指标
- `GET /metrics/workers`：节点维度指标

## 4. 错误码建议（一期）

- `1001`：参数校验失败
- `1002`：资源不存在
- `1003`：资源状态不允许当前操作
- `1004`：调度表达式非法
- `2001`：脚本包校验失败（SHA 不一致）
- `2002`：依赖安装失败
- `2003`：任务执行超时
- `2004`：Worker 不可用
- `2005`：Supervisor 启动失败
- `3001`：Mongo 写入失败
- `3002`：OSS 访问失败
- `9000`：系统内部异常

---

## 5. 一期落地建议（开发顺序）

1. 先实现：模板、脚本版本、依赖、任务、调度五大管理接口
2. 再实现：`internal` 调度通信接口 + Worker 心跳/拉取/上报
3. 最后实现：日志查询、监控指标、审计检索

该顺序可保证最短路径跑通「配置 -> 调度 -> 执行 -> 回传 -> 可观测」闭环。
